\section{Introduction et périmètre}
Cette application est une solution ERP web dédiée à la gestion multi-sites d'un réseau de clubs sportifs. Elle centralise la gestion structurelle (Ville $\rightarrow$ Salle $\rightarrow$ Groupe), la gestion des membres et du staff, les présences, la planification, la vie sportive (grades, événements, tests), ainsi que la finance (paiements, dépenses, reporting). Elle propose une interface bilingue (FR/AR) et une API pour les utilisateurs (\textit{mobile}).

\section{Acteurs, rôles et droits (RBAC)}
\subsection{Rôles principaux}
\begin{itemize}
    \item \textbf{Admin :} accès total (paramétrage, rôles, audit, finance, exports).
    \item \textbf{CityManager :} gestion des entités et personnes de sa ville (salles, managers, staff, membres).
    \item \textbf{GymManager :} gestion opérationnelle de sa salle (membres, sessions, cours, finance locale).
    \item \textbf{Coach :} accès pédagogique et opérationnel (cours/sessions, présences, suivi sportif).
    \item \textbf{User :} accès via API (authentification, profil, sessions, participation, présence).
\end{itemize}

\subsection{Gestion des rôles et permissions}
\begin{itemize}
    \item Création / modification / suppression de rôles.
    \item Clonage de rôles.
    \item Gestion des permissions (création / suppression).
    \item Liste des utilisateurs par rôle.
\end{itemize}

\section{Fonctionnalités transverses}
\subsection{Authentification et compte}
\begin{itemize}
    \item Connexion / déconnexion (web).
    \item Protection contre la force brute (\textit{rate limiting} du login).
    \item Réinitialisation de mot de passe (demande de lien, reset via token).
    \item Vérification d'email (web et API).
    \item Authentification API (Sanctum) avec jetons par appareil.
\end{itemize}

\subsection{Internationalisation et UI}
\begin{itemize}
    \item Bilinguisme FR/AR (bascule LTR/RTL).
    \item Mode sombre (préférences persistées).
    \item Recherche globale (\textit{Spotlight}) avec suggestions contextualisées selon le rôle.
    \item Interface responsive (mobile / tablette / desktop).
    \item Application installable (PWA : manifest + service worker + cache offline basique).
    \item Listes performantes côté serveur (\textit{DataTables}).
\end{itemize}

\subsection{Notifications et communication}
\begin{itemize}
    \item Notification email de bienvenue (asynchrone / queue).
    \item Notification email aux utilisateurs inactifs (commande planifiée).
    \item Emails système : vérification d'email, réinitialisation de mot de passe.
    \item Rappels de paiement par SMS (commande planifiée), avec mode \texttt{--dry-run} et délai \texttt{--days}.
    \item Fournisseur SMS configurable (multi-providers : Twilio / Orange) via une fabrique.
\end{itemize}

\subsection{Sécurité, conformité et audit}
\begin{itemize}
    \item Déconnexion forcée des utilisateurs bannis (\textit{logs-out-banned-user}).
    \item Détection et blocage des requêtes suspectes (\textit{honeypot}, outils d'attaque, patterns SQLi/XSS, scans automatisés).
    \item Nettoyage/sanitation des entrées (prévention XSS : \texttt{strip\_tags} récursif) et en-têtes de sécurité.
    \item Protection CSRF sur les routes web.
    \item En-têtes HTTP durcis : CSP, anti-clickjacking, anti-sniffing, Referrer-Policy, Permissions-Policy.
    \item Blocage automatique d'IP en cas de volume anormal ou scans (trop de requêtes / trop de 404), avec durée de blocage temporaire.
    \item Journalisation des actions (logs d'activité) : utilisateur, URL, méthode, IP, user-agent, payload filtré (sans mots de passe).
    \item Stockage privé des documents sensibles et téléchargements contrôlés.
\end{itemize}

\subsection{Documents, exports et reporting avancé}
\begin{itemize}
    \item Export PDF des feuilles de présence d'événements, avec support de l'arabe (mise en forme des glyphes).
    \item Génération de facture PDF pour les paiements (\textit{invoice}), avec numéro de facture unique lors du passage à \texttt{paid}.
    \item Exports de reporting en PDF et Excel, avec filtres multi-critères et sections activables/désactivables à l'export.
    \item Optimisations de performance sur les tableaux de bord : mise en cache des listes de référence et agrégats de rapports.
\end{itemize}

\section{Modules fonctionnels (liste exhaustive)}
\subsection{A. Gestion structurelle multi-sites}
\begin{itemize}
    \item \textbf{Villes (Cities) :} création, consultation, modification, suppression, restauration d'éléments supprimés.
    \item \textbf{Salles (Gyms) :} création, consultation, modification, suppression, rattachement à une ville.
    \item \textbf{Groupes :} création, modification, suppression, affectation de membres, action d'affectation en masse.
    \item \textbf{Catégories :} liste, création, modification, suppression, initialisation des catégories par \textit{seed}.
\end{itemize}

\subsection{B. Gestion des utilisateurs et membres}
\begin{itemize}
    \item \textbf{Annuaire (AllUsers) :} liste, recherche, création, consultation, modification, suppression.
    \item \textbf{Détails membre :} fiche complète (données personnelles, rattachements, informations sportives).
    \item \textbf{Affectation :} rattachement à une salle, affectation à un groupe.
    \item \textbf{Statut stage :} consultation et mise à jour du statut.
    \item \textbf{Documents :} ajout, suppression, téléchargement (documents membre).
    \item \textbf{Bannissement :} bannir, débannir, consulter la liste des utilisateurs bannis.
    \item \textbf{Rapport membre :} génération d'un rapport synthétique par membre.
\end{itemize}

\subsection{C. Gestion du staff et des managers}
\begin{itemize}
    \item \textbf{City Managers :} création, consultation, modification, suppression.
    \item \textbf{Gym Managers :} création, consultation, modification, suppression.
    \item \textbf{Coachs :} création, consultation, modification, suppression, fiche coach, rapport coach.
    \item \textbf{Staff :} création, consultation, modification, suppression (rôles hors \texttt{user}).
    \item \textbf{Modérateurs :} création, consultation, modification, suppression (avec contrôles RBAC).
\end{itemize}

\subsection{D. Planification, cours et séances}
\begin{itemize}
    \item \textbf{Cours :} liste, création, consultation, modification, suppression (version FR \texttt{/cours} et EN \texttt{/courses}).
    \item \textbf{Emplois du temps :} liste, création, modification, suppression (\texttt{Schedule}).
    \item \textbf{Séances d'entraînement :} liste, création, planification, consultation, modification, suppression (\texttt{TrainingSession}).
    \item \textbf{Packs d'entraînement :} liste, création, consultation, modification, suppression (\texttt{TrainingPackage}).
\end{itemize}

\subsection{E. Présences et assiduité}
\begin{itemize}
    \item \textbf{Présence (par séance) :} marquage, enregistrement, mise à jour AJAX, historique.
    \item \textbf{Présence par cours :} liste par cours, création, enregistrement, mise à jour AJAX, consultation par date, édition, suppression, statistiques.
    \item \textbf{Présence par catégorie :} création, enregistrement, mise à jour AJAX, consultation par date, édition, suppression, statistiques.
    \item \textbf{Présence coach :} saisie liée à un coach (route dédiée).
\end{itemize}

\subsection{F. Vie sportive : grades, notes, tests, combats, événements}
\begin{itemize}
    \item \textbf{Ceintures (Belts) :} liste, création, consultation, modification, suppression.
    \item \textbf{Passages de grades :} interface de grading, enregistrement des obtentions.
    \item \textbf{Notes de grade :} gestion des catégories, sous-catégories, items/notes (création/suppression).
    \item \textbf{Tests athlétiques :} gestion des tests (CRUD), insertion de valeurs par membre, rapport par membre, initialisation des tests par \textit{seed}.
    \item \textbf{Combats :} création d'oppositions, historique des combats.
    \item \textbf{Événements :} liste, détails, création, export PDF (feuille de présence), inscription d'un membre, gestion des inscriptions et marquage payé.
    \item \textbf{Participations :} liste, création, consultation, modification, suppression, gestion de \textit{waitlist}.
\end{itemize}

\subsection{G. Finance : paiements, dépenses, reporting}
\begin{itemize}
    \item \textbf{Stripe (achat de pack) :} écran de paiement, soumission, historique des achats.
    \item \textbf{Paiements par membre :} liste, création, modification, suppression, génération et téléchargement de facture PDF (\textit{invoice}) pour les paiements \texttt{paid}.
    \item \textbf{Dépenses :} liste, création, modification, suppression, rapport par coach.
    \item \textbf{Catégories de dépenses :} CRUD catégories.
    \item \textbf{Sous-catégories de dépenses :} CRUD sous-catégories.
    \item \textbf{Rapports :} tableau de bord, filtres (mois, ville, salle, coach, cours, statuts, montants), exports PDF/Excel.
    \item \textbf{Revenus :} ventilation par source (packs, inscriptions événements, examens de ceinture).
    \item \textbf{Synthèse :} total membres, actifs/bannis, volumes salles/cours/événements, top coachs, top salles.
\end{itemize}

\subsection{H. Administration et paramétrage}
\begin{itemize}
    \item \textbf{Accueil (welcome) :} page d'entrée staff avec raccourcis et accès rapide.
    \item \textbf{Paramètres globaux :} consultation et mise à jour.
    \item \textbf{Logs d'activité :} liste et consultation détaillée d'entrées.
    \item \textbf{Pages système :} page vide/maintenance et page non autorisée (\texttt{/unAuth}).
    \item \textbf{Profil admin :} consultation et modification du profil (routes dédiées).
\end{itemize}

\section{API (mobile / intégrations)}
\subsection{Authentification et compte}
\begin{itemize}
    \item \textbf{Auth :} \texttt{POST /api/signin}, \texttt{POST /api/signup}, \texttt{GET /api/logout}.
    \item \textbf{Profil :} \texttt{POST /api/update\_profile}.
    \item \textbf{Email :} renvoi de notification de vérification et vérification via lien.
    \item \textbf{Jetons :} \texttt{POST /api/sanctum/token}.
\end{itemize}

\subsection{Séances, packs, cours, participations, présences}
\begin{itemize}
    \item \textbf{Séances :} listing, détails, sessions restantes, présence à une session (\texttt{attendSession}).
    \item \textbf{Packs :} listing et détails.
    \item \textbf{CRUD :} \texttt{courses}, \texttt{participations}, \texttt{attendances} via ressources REST.
    \item \textbf{Filtrage :} participations par utilisateur et par cours.
\end{itemize}

\section{Automatisation et maintenance}
\begin{itemize}
    \item Commande \texttt{create:admin} : création d'un compte admin via CLI.
    \item Commande \texttt{notify:users-not-logged-in-for-month} : notification des utilisateurs inactifs.
    \item Commande \texttt{payments:remind} : rappels SMS sur paiements à échéance (options \texttt{--days} et \texttt{--dry-run}).
    \item Exécution planifiée (scheduler) pour l'automatisation des notifications.
\end{itemize}
